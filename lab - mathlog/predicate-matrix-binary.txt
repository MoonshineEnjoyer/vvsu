import heapq
from collections import Counter

# ЗАДАНИЕ 1

class HuffNode:
    def __init__(self, char, freq):
        self.char = char
        self.freq = freq
        self.left = None
        self.right = None

    def __lt__(self, other):
        return self.freq < other.freq  # для heapq


def build_huffman_tree(text):
    frequencies = Counter(text)  # счёт частоты
    heap = [HuffNode(char, freq) for char, freq in frequencies.items()]  # список узлов каждого символа и его частоты
    heapq.heapify(heap)  # преобразование, O(n)
    while len(heap) > 1:  # >1 узла => объединяем с наимен. част.
        a = heapq.heappop(heap)  # мин.ч
        b = heapq.heappop(heap)  # 2 мин.ч
        merged = HuffNode(None, a.freq + b.freq)
        merged.left = a
        merged.right = b
        heapq.heappush(heap, merged)  # новый узел - обратно в очер.
    return heap[0]  # корень дерева


def build_codes(node, prefix="", codes=None):  # получаем коды Хаффмана для каждого символа, рекурсия
    if codes is None:
        codes = {}
    if node.char is not None:  #  # есть символ? (является листом?)
        codes[node.char] = prefix or "0"  # сохраняем код для символа в словаре codes
        return codes
    build_codes(node.left, prefix + "0", codes)  # идём влево, добавляя 0 к текущему коду (рек.)
    build_codes(node.right, prefix + "1", codes)  # идём вправо, добавляя 1 к текущему коду (рек.)
    return codes


def encode(text, codes):  # кодируем
    return "".join(codes[char] for char in text)  # заменяем каждый символ на его код и объединяем

# ДЕКОДИРОВАНИЕ ПО ДЕРЕВУ
def decode_with_tree(encoded_text, root):  # проходим по дереву в соотв. с битами закод. строки
    decoded_text = []
    current_huffnode = root
    for bit in encoded_text:
        if bit == "0":  # бит = 0 => левый потомок текущего узла
            current_huffnode = (current_huffnode.left)  # бит = 1 => правый потомок текущего узла
        else:  # бит = 1 => правый потомок текущего узла
            current_huffnode = current_huffnode.right
        if current_huffnode.char is not None:  # есть символ? (является листом?)
            decoded_text.append(current_huffnode.char)  # да => символ в decoded_text
            current_huffnode = root  # возвращаемся, декодируем след символ
    return "".join(decoded_text)  # объединяем декодированные символы


# ДЕКОДИРОВАНИЕ ПО ТАБЛИЦЕ
def decode_with_table(encoded_text, codes):  # инвертируем таблицу для получения символов по кодам: код -> символ
    rev_codes = {v: k for k, v in codes.items()}  # ключи - коды, значения - символы
    decoded_text = ""
    temp_code = ""  # будем проверять текущий код на наличие в обр. таблице
    for bit in encoded_text:
        temp_code += bit  # бит + к текущему коду
        if temp_code in rev_codes:  # проверяем текущий код на наличие в обр. таблице
            decoded_text += rev_codes[temp_code]  # код найден => плюс символ
            temp_code = ""
    return decoded_text

# 1. Часто встречающиеся символы располагаются ближе к корню дерева и получают короткие коды. Редкие символы вытесняются на нижние уровни, что суммарно экономит биты при кодировании текста.
# 2. Ни одно кодовое слово не является началом другого (01 - значит нет 011, 010 и т.д.). Гарантирует однозначное декодирование потока бит слева направо без использования разделителей между символами.
# 3. O(n), n - длина строки. По готовому дереву за (O(K)) шагов строится хэш-таблица «символ -> битовый код». Затем исходная строка обходится ровно один раз, и каждый символ заменяется готовой битовой цепочкой за (O(1)).
# 4. Равномерное распределение частот символов (особенно если их количество — степень двойки). Невозможно сократить длину кода для одних символов за счет других, если у всех знаков одинаковая вероятность появления.

# 1. Чтение битов потока как команд навигации: 0 — шаг влево, 1 — шаг вправо. Стартуем из корня. Двигаемся по битам вниз до тех пор, пока не упремся в лист. Символ из листа пишется в ответ, а указатель текущего узла мгновенно сбрасывается обратно в корень для чтения следующего бита.
# 2. Листья дерева Хаффмана не имеют потомков. Как только алгоритм доходит до листа, путь изолируется — этот код физически не может быть продолжен во что-то другое. Это исключает необходимость заглядывать вперед.
# 3. Полное разрушение («рассинхронизация») всего последующего текста. Одиночная ошибка не просто исказит один символ, а заставит алгоритм завершить шаг в листе раньше или позже положенного. Из-за этого границы всех следующих символов сместятся, и поток превратится в хаотичный мусор.
