class PrefixCalculator:
    def __init__(self):
        # Операторы и их приоритеты
        self.precedence = {'+': 1, '-': 1, '*': 2, '/': 2, '**': 3}

    def _tokenize(self, expr):
        """Ручной токенизатор (поддержка чисел, float, операторов и скобок)"""
        tokens = []
        i = 0
        while i < len(expr):
            char = expr[i]
            if char.isspace():
                i += 1
                continue
            if char.isdigit() or char == '.':
                num = ""
                while i < len(expr) and (expr[i].isdigit() or expr[i] == '.'):
                    num += expr[i]
                    i += 1
                tokens.append(num)
                continue
            if expr[i:i+2] == '**':
                tokens.append('**')
                i += 2
                continue
            tokens.append(char)
            i += 1
        return tokens

    def infix_to_postfix(self, expression):
        """Вспомогательный метод для таблицы (пункт 4)"""
        tokens = self._tokenize(expression)
        output, stack = [], []
        for token in tokens:
            if token.replace('.', '', 1).isdigit():
                output.append(token)
            elif token == '(':
                stack.append(token)
            elif token == ')':
                while stack and stack[-1] != '(':
                    output.append(stack.pop())
                if stack: stack.pop()
            else:
                while stack and stack[-1] != '(' and self.precedence.get(stack[-1], 0) >= self.precedence.get(token, 0):
                    if token == '**' and stack[-1] == '**': break
                    output.append(stack.pop())
                stack.append(token)
        while stack: output.append(stack.pop())
        return " ".join(output)

    def infix_to_prefix(self, expression):
        """Пункт 1: Инфикс -> Префикс (через инверсию)"""
        tokens = self._tokenize(expression)[::-1]
        for i in range(len(tokens)):
            if tokens[i] == '(': tokens[i] = ')'
            elif tokens[i] == ')': tokens[i] = '('
        
        output, stack = [], []
        for token in tokens:
            if token.replace('.', '', 1).isdigit():
                output.append(token)
            elif token == '(':
                stack.append(token)
            elif token == ')':
                while stack and stack[-1] != '(':
                    output.append(stack.pop())
                if stack: stack.pop()
            else:
                # Для префикса при равенстве приоритетов НЕ выталкиваем
                while stack and stack[-1] != '(' and self.precedence.get(stack[-1], 0) > self.precedence.get(token, 0):
                    output.append(stack.pop())
                stack.append(token)
        while stack: output.append(stack.pop())
        return " ".join(output[::-1])

    def evaluate_prefix(self, prefix_expr):
        """Пункт 2: Вычисление префикса с обработкой ошибок"""
        tokens = prefix_expr.split()
        stack = []
        try:
            for i in range(len(tokens) - 1, -1, -1):
                token = tokens[i]
                if token.replace('.', '', 1).isdigit():
                    stack.append(float(token))
                else:
                    if len(stack) < 2: raise ValueError("Недостаточно операндов")
                    a, b = stack.pop(), stack.pop()
                    if token == '+': stack.append(a + b)
                    elif token == '-': stack.append(a - b)
                    elif token == '*': stack.append(a * b)
                    elif token == '/':
                        if b == 0: raise ZeroDivisionError("Деление на ноль")
                        stack.append(a / b)
                    elif token == '**': stack.append(a ** b)
            if len(stack) != 1: raise ValueError("Ошибка структуры выражения")
            return stack[0]
        except Exception as e:
            return f"Ошибка: {e}"

    def evaluate_postfix(self, postfix_expr):
        """Вспомогательный метод вычисления постфикса (для пункта 5)"""
        tokens = postfix_expr.split()
        stack = []
        try:
            for token in tokens:
                if token.replace('.', '', 1).isdigit():
                    stack.append(float(token))
                else:
                    b, a = stack.pop(), stack.pop()
                    if token == '+': stack.append(a + b)
                    elif token == '-': stack.append(a - b)
                    elif token == '*': stack.append(a * b)
                    elif token == '/':
                        if b == 0: raise ZeroDivisionError
                        stack.append(a / b)
                    elif token == '**': stack.append(a ** b)
            return stack[0]
        except: return None

def compare_evaluate(expression):
    """Пункт 5: Сравнение двух методов вычисления"""
    calc = PrefixCalculator()
    
    prefix = calc.infix_to_prefix(expression)
    postfix = calc.infix_to_postfix(expression)
    
    res_pre = calc.evaluate_prefix(prefix)
    res_post = calc.evaluate_postfix(postfix)
    
    # Проверка на совпадение (учитываем, что res_pre может быть строкой-ошибкой)
    if isinstance(res_pre, (int, float)) and res_post is not None:
        match = abs(res_pre - res_post) < 1e-9
    else:
        match = False
        
    return res_pre, res_post, match

calc = PrefixCalculator()

# Пункт 4: Вывод таблицы
print("№4. ТАБЛИЦА ВЫРАЖЕНИЙ")
test_exprs = ["3 - 4 * 2", "(3 + 4) * 2", "10 - 2 * (3 + 1)"]
header = f"{'Infix':<20} | {'Postfix':<20} | {'Prefix':<20}"
print(header)
print("-" * len(header))
for e in test_exprs:
    print(f"{e:<20} | {calc.infix_to_postfix(e):<20} | {calc.infix_to_prefix(e):<20}")

# Пункт 5 + Ввод пользователя
print("\n№5. СРАВНЕНИЕ И ПОЛЬЗОВАТЕЛЬСКИЙ ВВОД")
user_input = input("Введите инфиксное выражение: ")
res_p, res_po, is_ok = compare_evaluate(user_input)

print(f"Результат (Prefix): {res_p}")
print(f"Результат (Postfix): {res_po}")
print(f"Результаты совпадают: {is_ok}")
