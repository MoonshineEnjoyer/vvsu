class Car:
    # Общий атрибут для всех объектов класса
    wheels = 4

    def __init__(self, brand, model, year, color, mileage, fuel_level):
        # 6 атрибутов, описывающих объект
        self.brand = brand
        self.model = model
        self.year = year
        self.color = color
        self.mileage = mileage
        self.fuel_level = fuel_level

    # 1. Метод возвращает строку с краткой инфой
    def get_info(self):
        return f"{self.brand} {self.model} ({self.year})"

    # 2. Метод возвращает полный статус
    def get_status(self):
        return f"Цвет: {self.color}, Пробег: {self.mileage}км, Топливо: {self.fuel_level}%"

    # 3. Метод меняет значение атрибута (пробег)
    def drive(self, distance):
        self.mileage += distance
        return f"Проехали {distance} км. Новый пробег: {self.mileage}"

    # 4. Метод меняет значение атрибута (заправка)
    def refuel(self, amount):
        self.fuel_level = min(100, self.fuel_level + amount)
        return f"Заправились на {amount}%. Уровень топлива: {self.fuel_level}%"

def memory(obj, seen=None):
    if seen is None:
        seen = set()
    if id(obj) in seen:
        return 0
    seen.add(id(obj))
    size = sys.getsizeof(obj)
    if isinstance(obj, (list, tuple, dict, set)):
        items = obj.values() if isinstance(obj, dict) else obj
        for item in items:
            size += memory(item, seen)
    return size

def find_max_brute_force(arr):
    n = len(arr)
    for i in range(n):
        is_max = True
        for j in range(n):
            if arr[j] > arr[i]:
                is_max = False
                break
        if is_max:
            return arr[i]

def find_max_greedy(arr):
    max_val = arr[0]
    for x in arr:
        if x > max_val:
            max_val = x
    return max_val

# 3 задание, Divide & Conquer
def find_max_divide_conquer(arr, low, high):
    if low == high:
        return arr[low]
    mid = (low + high) // 2
    left_max = find_max_divide_conquer(arr, low, mid)
    right_max = find_max_divide_conquer(arr, mid + 1, high)
    return left_max if left_max > right_max else right_max

# 3 задание, DP
def find_max_dp(arr):
    n = len(arr)
    dp = [0] * n
    dp[0] = arr[0]
    for i in range(1, n):
        dp[i] = max(dp[i - 1], arr[i])
    return dp[n - 1]

import time
import random

# 1. Базовая версия
def bubble_sort_basic(arr):
    n = len(arr)
    for i in range(n):
        for j in range(0, n - i - 1):
            if arr[j] > arr[j + 1]:
                arr[j], arr[j + 1] = arr[j + 1], arr[j]
    return arr

# 2. С флагом раннего выхода
def bubble_sort_flag(arr):
    n = len(arr)
    for i in range(n):
        swapped = False
        for j in range(0, n - i - 1):
            if arr[j] > arr[j + 1]:
                arr[j], arr[j + 1] = arr[j + 1], arr[j]
                swapped = True
        if not swapped: # Если обменов не было — массив отсортирован
            break
    return arr

# 3. С ограничением границы последнего обмена
def bubble_sort_limit(arr):
    n = len(arr)
    while n > 1:
        new_n = 0
        for i in range(n - 1):
            if arr[i] > arr[i + 1]:
                arr[i], arr[i + 1] = arr[i + 1], arr[i]
                new_n = i + 1 # Запоминаем последнее место обмена
        n = new_n # Дальше этой границы всё уже отсортировано
    return arr

# Функция для тестирования
def test_sorting():
    n = 5000
    test_cases = {
        "Отсортированный": list(range(n)),
        "Случайный": [random.randint(0, n) for _ in range(n)],
        "Обратный": list(range(n, 0, -1))
    }
    
    sorts = [
        ("Базовая", bubble_sort_basic),
        ("С флагом", bubble_sort_flag),
        ("С границей", bubble_sort_limit)
    ]

    for case_name, data in test_cases.items():
        print(f"\n--- Массив: {case_name} ---")
        for sort_name, func in sorts:
            arr_copy = data.copy()
            start = time.perf_counter()
            func(arr_copy)
            end = time.perf_counter()
            print(f"{sort_name:12}: {end - start:.4f} сек")

test_sorting()

# 1. Линейный поиск
def linear_search(arr, target):
    comparisons = 0
    for i in range(len(arr)):
        comparisons += 1
        if arr[i] == target:
            return True, comparisons
    return False, comparisons

# 2. Бинарный поиск (только для отсортированного массива!)
def binary_search(arr, target):
    comparisons = 0
    low = 0
    high = len(arr) - 1
    
    while low <= high:
        comparisons += 1
        mid = (low + high) // 2
        if arr[mid] == target:
            return True, comparisons
        elif arr[mid] < target:
            low = mid + 1
        else:
            high = mid - 1
    return False, comparisons
