while True:
    password = input("\nВведите пароль: ")
    errors = []

    # 1. Проверка длины
    if len(password) < 8:
        errors.append("Пароль слишком короткий (минимум 8 символов)")

    # 2. Проверка заглавной латиницы (диапазон A-Z)
    if not any('A' <= char <= 'Z' for char in password):
        errors.append("Нет заглавных латинских букв (A-Z)")

    # 3. Проверка строчной латиницы (диапазон a-z)
    if not any('a' <= char <= 'z' for char in password):
        errors.append("Нет строчных латинских букв (a-z)")

    # 4. Проверка цифр
    if not any('0' <= char <= '9' for char in password):
        errors.append("Нет цифр")

    # 5. Проверка на пробелы
    if " " in password:
        errors.append("Пароль содержит пробелы")

    # Вывод результата
    if not errors:
        print("✅ Пароль надежный!")
    else:
        print("❌ Ошибки:")
        for error in errors:
            print(f"- {error}")

    # Повтор
    choice = input("\nХотите попробовать еще раз? (да/нет): ").strip().lower()
    if choice != 'да':
        print("До свидания!")
        break

# 1. Создание каталога (список словарей)

while True:
    # 2. Запрос имени автора
    search_author = input("\nВведите имя автора для поиска: ").strip().lower()
    
    # 3. Поиск книг (без учета регистра)
    results = []
    for book in catalog:
        if search_author == book["author"].lower():
            results.append(book)
            
    # 4. Вывод результатов
    if results:
        print(f"\nНайдено книг: {len(results)}")
        for i, book in enumerate(results, 1):
            print(f"{i}. '{book['title']}', Автор: {book['author']}, Год: {book['year']}, Жанр: {book['genre']}")
    else:
        print("❌ Книги автора не найдены")

    # 5. Предложение продолжить
    choice = input("\nХотите повторить поиск? (да/нет): ").strip().lower()

        print("Программа завершена. Всего доброго!")
