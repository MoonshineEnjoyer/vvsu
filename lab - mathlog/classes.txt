class PrefixCalculator:
    def __init__(self):
        # Словарь приоритетов
        self.precedence = {'+': 1, '-': 1, '*': 2, '/': 2, '**': 3}

    def _tokenize(self, expr):
        """Ручной разбор строки на числа, операторы и скобки."""
        tokens = []
        i = 0
        while i < len(expr):
            char = expr[i]
            if char.isspace():
                i += 1
                continue
            # Сборка числа (целого или вещественного)
            if char.isdigit() or char == '.':
                num = ""
                while i < len(expr) and (expr[i].isdigit() or expr[i] == '.'):
                    num += expr[i]
                    i += 1
                tokens.append(num)
                continue
            # Сборка оператора степени
            if expr[i:i+2] == '**':
                tokens.append('**')
                i += 2
                continue
            # Остальные одиночные символы
            tokens.append(char)
            i += 1
        return tokens

    def infix_to_prefix(self, expression):
        """Перевод инфиксной записи в префиксную через инверсию."""
        tokens = self._tokenize(expression)
        
        # 1. Разворачиваем список токенов
        tokens = tokens[::-1]
        
        # 2. Инвертируем скобки
        for i in range(len(tokens)):
            if tokens[i] == '(': tokens[i] = ')'
            elif tokens[i] == ')': tokens[i] = '('
        
        # 3. Алгоритм сортировочной станции
        output = []
        stack = []
        for token in tokens:
            # Если токен - число
            if token.replace('.', '', 1).isdigit():
                output.append(token)
            elif token == '(':
                stack.append(token)
            elif token == ')':
                while stack and stack[-1] != '(':
                    output.append(stack.pop())
                if stack: stack.pop()
            else:
                # Для префиксной записи: строгое '>' при сравнении приоритетов
                while (stack and stack[-1] != '(' and 
                       self.precedence.get(stack[-1], 0) > self.precedence.get(token, 0)):
                    output.append(stack.pop())
                stack.append(token)
        
        while stack:
            output.append(stack.pop())
            
        # 4. Разворачиваем результат обратно и склеиваем
        prefix_list = output[::-1]
        return " ".join(prefix_list)

    def evaluate_prefix(self, prefix_expr):
        """Вычисление префиксного выражения (справа налево)."""
        tokens = prefix_expr.split()
        stack = []
        
        # Проход по токенам в обратном порядке
        for i in range(len(tokens) - 1, -1, -1):
            token = tokens[i]
            if token.replace('.', '', 1).isdigit():
                stack.append(float(token))
            else:
                if len(stack) < 2:
                    return "Ошибка: Некорректная префиксная запись"
                # Извлекаем операнды
                a = stack.pop()
                b = stack.pop()
                
                if token == '+': stack.append(a + b)
                elif token == '-': stack.append(a - b)
                elif token == '*': stack.append(a * b)
                elif token == '/':
                    if b == 0: return "Ошибка: Деление на ноль"
                    stack.append(a / b)
                elif token == '**': stack.append(a ** b)
        
        return stack[0] if stack else "Ошибка"

# --- Основной блок выполнения ---

calc = PrefixCalculator()

# Запрос ввода у пользователя
user_expression = input("Введите инфиксное выражение (например, 10 - 2 * (3 + 1)): ")

# Выполнение задач
prefix_result = calc.infix_to_prefix(user_expression)
final_value = calc.evaluate_prefix(prefix_result)

# Вывод результатов
print("-" * 30)
print(f"Инфиксная запись: {user_expression}")
print(f"Префиксная запись: {prefix_result}")
print(f"Результат вычисления: {final_value}")
