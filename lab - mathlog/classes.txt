class PrefixCalculator:
    def __init__(self):
        # Приоритеты операций (~ - унарный минус)
        self.precedence = {'+': 1, '-': 1, '*': 2, '/': 2, '**': 3, '~': 4}

    def _tokenize(self, expr):
        """Ручной токенизатор с поддержкой унарного минуса и вещественных чисел."""
        tokens = []
        i = 0
        while i < len(expr):
            char = expr[i]
            if char.isspace():
                i += 1
                continue
            
            # Логика унарного минуса
            if char == '-':
                if i == 0 or tokens[-1] in ('+', '-', '*', '/', '(', '**', '~'):
                    tokens.append('~')
                    i += 1
                    continue
            
            # Числа (целые и float)
            if char.isdigit() or char == '.':
                num = ""
                while i < len(expr) and (expr[i].isdigit() or expr[i] == '.'):
                    num += expr[i]
                    i += 1
                tokens.append(num)
                continue
            
            # Возведение в степень
            if expr[i:i+2] == '**':
                tokens.append('**')
                i += 2
                continue
            
            # Прочие символы (операторы и скобки)
            tokens.append(char)
            i += 1
        return tokens

    def infix_to_postfix(self, expression):
        """Перевод в постфикс (для таблицы по п.4)."""
        tokens = self._tokenize(expression)
        output, stack = [], []
        for token in tokens:
            if token.replace('.', '', 1).isdigit():
                output.append(token)
            elif token == '(':
                stack.append(token)
            elif token == ')':
                while stack and stack[-1] != '(':
                    output.append(stack.pop())
                if stack: stack.pop()
            else:
                while stack and stack[-1] != '(' and self.precedence.get(stack[-1], 0) >= self.precedence.get(token, 0):
                    if token in ('**', '~') and stack[-1] == token: break
                    output.append(stack.pop())
                stack.append(token)
        while stack: output.append(stack.pop())
        return " ".join(output)

    def infix_to_prefix(self, expression):
        """Пункт 1: Инфикс -> Префикс (через инверсию по п.3)."""
        tokens = self._tokenize(expression)[::-1]
        # Инвертируем скобки
        for i in range(len(tokens)):
            if tokens[i] == '(': tokens[i] = ')'
            elif tokens[i] == ')': tokens[i] = '('
        
        output, stack = [], []
        for token in tokens:
            if token.replace('.', '', 1).isdigit():
                output.append(token)
            elif token == '(':
                stack.append(token)
            elif token == ')':
                while stack and stack[-1] != '(':
                    output.append(stack.pop())
                if stack: stack.pop()
            else:
                # В префиксе при равенстве приоритетов не выталкиваем
                while stack and stack[-1] != '(' and self.precedence.get(stack[-1], 0) > self.precedence.get(token, 0):
                    output.append(stack.pop())
                stack.append(token)
        while stack: output.append(stack.pop())
        return " ".join(output[::-1])

    def evaluate_prefix(self, prefix_expr):
        """Пункт 2: Вычисление префикса (вещественные числа + ошибки)."""
        tokens = prefix_expr.split()
        stack = []
        try:
            for i in range(len(tokens) - 1, -1, -1):
                token = tokens[i]
                if token.replace('.', '', 1).isdigit():
                    stack.append(float(token))
                elif token == '~':
                    if not stack: raise ValueError("Ошибка операндов")
                    stack.append(-stack.pop())
                else:
                    if len(stack) < 2: raise ValueError("Некорректная структура")
                    a, b = stack.pop(), stack.pop()
                    if token == '+': stack.append(a + b)
                    elif token == '-': stack.append(a - b)
                    elif token == '*': stack.append(a * b)
                    elif token == '/':
                        if b == 0: raise ZeroDivisionError("Деление на ноль")
                        stack.append(a / b)
                    elif token == '**': stack.append(a ** b)
            
            if len(stack) != 1: raise ValueError("Лишние операнды")
            return float(stack[0])
        except Exception as e:
            return f"Ошибка: {e}"

    def evaluate_postfix(self, postfix_expr):
        """Вычисление постфикса для сравнения по п.5."""
        tokens = postfix_expr.split()
        stack = []
        try:
            for token in tokens:
                if token.replace('.', '', 1).isdigit():
                    stack.append(float(token))
                elif token == '~':
                    stack.append(-stack.pop())
                else:
                    b, a = stack.pop(), stack.pop()
                    if token == '+': stack.append(a + b)
                    elif token == '-': stack.append(a - b)
                    elif token == '*': stack.append(a * b)
                    elif token == '/': stack.append(a / b)
                    elif token == '**': stack.append(a ** b)
            return float(stack[0])
        except: return None

def compare_evaluate(expression):
    """Пункт 5: Сравнение двух подходов."""
    calc = PrefixCalculator()
    prefix = calc.infix_to_prefix(expression)
    postfix = calc.infix_to_postfix(expression)
    
    res_pre = calc.evaluate_prefix(prefix)
    res_post = calc.evaluate_postfix(postfix)
    
    match = False
    if isinstance(res_pre, float) and res_post is not None:
        match = abs(res_pre - res_post) < 1e-9
        
    return prefix, postfix, res_pre, res_post, match

class PrefixCalculator:
    def __init__(self):
        # Приоритеты: чем выше число, тем быстрее выполняется операция
        # '~' — это внутреннее обозначение унарного минуса (самый высокий приоритет)
        self.precedence = {'+': 1, '-': 1, '*': 2, '/': 2, '**': 3, '~': 4}

    def _tokenize(self, expr):
        """Разбор строки на значимые части (токены) вручную."""
        tokens = []
        i = 0
        while i < len(expr):
            char = expr[i]
            if char.isspace(): # Игнорируем пробелы
                i += 1
                continue
            
            # Определение унарного минуса (если минус в начале или после оператора)
            if char == '-':
                if i == 0 or tokens[-1] in ('+', '-', '*', '/', '(', '**', '~'):
                    tokens.append('~')
                    i += 1
                    continue
            
            # Сборка числа (целого или с точкой)
            if char.isdigit() or char == '.':
                num = ""
                while i < len(expr) and (expr[i].isdigit() or expr[i] == '.'):
                    num += expr[i]
                    i += 1
                tokens.append(num)
                continue
            
            # Поддержка оператора степени (двойной символ)
            if expr[i:i+2] == '**':
                tokens.append('**')
                i += 2
                continue
            
            tokens.append(char)
            i += 1
        return tokens

    def infix_to_postfix(self, expression):
        """Классический алгоритм 'Сортировочная станция' (Infix -> Postfix)."""
        tokens = self._tokenize(expression)
        output, stack = [], []
        for token in tokens:
            if token.replace('.', '', 1).isdigit(): # Если число — сразу в ответ
                output.append(token)
            elif token == '(': # Открывающая скобка всегда в стек
                stack.append(token)
            elif token == ')': # Закрывающая — выталкиваем всё до '('
                while stack and stack[-1] != '(':
                    output.append(stack.pop())
                if stack: stack.pop() # Удаляем саму '('
            else:
                # Если пришел оператор, выталкиваем из стека те, что сильнее или равны
                while stack and stack[-1] != '(' and self.precedence.get(stack[-1], 0) >= self.precedence.get(token, 0):
                    # Исключение для правоассоциативных (**) — они ложатся сверху при равенстве
                    if token in ('**', '~') and stack[-1] == token: break
                    output.append(stack.pop())
                stack.append(token)
        while stack: output.append(stack.pop())
        return " ".join(output)

    def infix_to_prefix(self, expression):
        """Алгоритм перевода в префикс через двойную инверсию (согласно П.3 ТЗ)."""
        # 1. Инвертируем входные токены
        tokens = self._tokenize(expression)[::-1]
        
        # 2. Меняем скобки местами (открывающая становится закрывающей)
        for i in range(len(tokens)):
            if tokens[i] == '(': tokens[i] = ')'
            elif tokens[i] == ')': tokens[i] = '('
        
        output, stack = [], []
        for token in tokens:
            if token.replace('.', '', 1).isdigit():
                output.append(token)
            elif token == '(':
                stack.append(token)
            elif token == ')':
                while stack and stack[-1] != '(':
                    output.append(stack.pop())
                if stack: stack.pop()
            else:
                # Важное отличие для префикса: выталкиваем только если строго БОЛЬШЕ (>)
                while stack and stack[-1] != '(' and self.precedence.get(stack[-1], 0) > self.precedence.get(token, 0):
                    output.append(stack.pop())
                stack.append(token)
        
        while stack: output.append(stack.pop())
        # 3. Инвертируем результат обратно
        return " ".join(output[::-1])

    def evaluate_prefix(self, prefix_expr):
        """Вычисление префикса (Пункт 2: поддержка float и обработка ошибок)."""
        tokens = prefix_expr.split()
        stack = []
        try:
            # Читаем ПРЕФИКС справа налево
            for i in range(len(tokens) - 1, -1, -1):
                token = tokens[i]
                if token.replace('.', '', 1).isdigit():
                    stack.append(float(token))
                elif token == '~': # Унарный минус меняет знак одного числа
                    stack.append(-stack.pop())
                else:
                    # Извлекаем два числа для бинарного оператора
                    if len(stack) < 2: raise ValueError("Неверная структура")
                    a, b = stack.pop(), stack.pop()
                    if token == '+': stack.append(a + b)
                    elif token == '-': stack.append(a - b)
                    elif token == '*': stack.append(a * b)
                    elif token == '/':
                        if b == 0: raise ZeroDivisionError("Деление на ноль")
                        stack.append(a / b)
                    elif token == '**': stack.append(a ** b)
            return float(stack[0])
        except Exception as e:
            return f"Ошибка: {e}"

    def evaluate_postfix(self, postfix_expr):
        """Вычисление постфикса (для сравнения в П.5)."""
        tokens = postfix_expr.split()
        stack = []
        try:
            for token in tokens:
                if token.replace('.', '', 1).isdigit():
                    stack.append(float(token))
                elif token == '~':
                    stack.append(-stack.pop())
                else:
                    b, a = stack.pop(), stack.pop() # Важен порядок: b - верхний, a - под ним
                    if token == '+': stack.append(a + b)
                    elif token == '-': stack.append(a - b)
                    elif token == '*': stack.append(a * b)
                    elif token == '/': stack.append(a / b)
                    elif token == '**': stack.append(a ** b)
            return float(stack[0])
        except: return None

def compare_evaluate(expression):
    """Сравнение двух методов (Пункт 5)."""
    calc = PrefixCalculator()
    prefix = calc.infix_to_prefix(expression)
    postfix = calc.infix_to_postfix(expression)
    
    res_pre = calc.evaluate_prefix(prefix)
    res_post = calc.evaluate_postfix(postfix)
    
    match = False
    if isinstance(res_pre, float) and res_post is not None:
        # Сравнение float через допустимую погрешность (эпсилон)
        match = abs(res_pre - res_post) < 1e-9
        
    return prefix, postfix, res_pre, res_post, match

import math
import timeit

# Создаём узел
class Node:
    def __init__(self, value):
        self.value = value
        self.next = None

# Создание стека (связной список, LIFO)
class Stack:
    def __init__(self):
        self.top = None
        self._size = 0

    def push(self, value):
        new_node = Node(value)
        new_node.next = self.top
        self.top = new_node
        self._size += 1

    def pop(self):
        if self.is_empty():
            raise IndexError("Pop из пустого стека")
        val = self.top.value
        self.top = self.top.next
        self._size -= 1
        return val

    def peek(self):
        return self.top.value if self.top else None

    def is_empty(self):
        return self.top is None

# Очередь, связной список LIFO O(1)
class Queue:
    def __init__(self):
        self.head = None
        self.tail = None
        self._size = 0

    def enqueue(self, value):
        new_node = Node(value)
        if self.tail:
            self.tail.next = new_node
        self.tail = new_node
        if not self.head:
            self.head = new_node
        self._size += 1

    def dequeue(self):
        if self.is_empty():
            raise IndexError("Dequeue из пустой очереди")
        val = self.head.value
        self.head = self.head.next
        if not self.head:
            self.tail = None
        self._size -= 1
        return val

    def is_empty(self):
        return self.head is None

class LinkedListCalculator:
    def __init__(self):
        self.ops = {
            '+': (1, 'L'), '-': (1, 'L'), # Сложение
            '*': (2, 'L'), '/': (2, 'L'), # Умножение
            '**': (3, 'R'), # Возведение в степень
            'u-': (4, 'R')  # Унарный минус
        }
        self.funcs = {'sin', 'cos', 'tg', 'ctg', 'sqrt'}

    def _tokenize(self, expr):
        tokens = Queue()
        expr = expr.replace(' ', '').replace(',', '.')
        i = 0
        prev_token = None
        while i < len(expr):
            char = expr[i]
            # Проверка ввода чисел
            if char.isdigit() or char == '.':
                num = ""
                while i < len(expr) and (expr[i].isdigit() or expr[i] == '.'):
                    num += expr[i]
                    i += 1
                tokens.enqueue(num)
                prev_token = num
                continue
            if char == '-':
                if prev_token is None or prev_token == '(' or prev_token in self.ops:
                    token = 'u-'
                else:
                    token = '-'
                tokens.enqueue(token)
                prev_token = token
                i += 1
            # Проверка ввода букв (т.е. функций)
            elif char.isalpha():
                fn = ""
                while i < len(expr) and expr[i].isalpha():
                    fn += expr[i]
                    i += 1
                tokens.enqueue(fn)
                prev_token = fn
            # Проверка на возведение в степень
            elif char == '*' and i + 1 < len(expr) and expr[i+1] == '*':
                tokens.enqueue('**')
                i += 2
            # Остальные символы (+, -, *, /, (, ))
            else:
                tokens.enqueue(char)
                prev_token = char
                i += 1
        return tokens

    def to_postfix(self, expression):
        tokens = self._tokenize(expression)
        output = Queue()
        stack = Stack()

        while not tokens.is_empty():
            t = tokens.dequeue()
            if t[0].isdigit() or (len(t) > 1 and t[1].isdigit()):
                output.enqueue(t)
            elif t in self.funcs:
                stack.push(t)
            elif t == '(':
                stack.push(t)
            elif t == ')':
                while not stack.is_empty() and stack.peek() != '(':
                    output.enqueue(stack.pop())
                stack.pop() # убираем '('
                if not stack.is_empty() and stack.peek() in self.funcs:
                    output.enqueue(stack.pop())
            elif t in self.ops:
                p1, a1 = self.ops[t]
                while not stack.is_empty() and stack.peek() in self.ops:
                    p2, _ = self.ops[stack.peek()]
                    if (a1 == 'L' and p1 <= p2) or (a1 == 'R' and p1 < p2):
                        output.enqueue(stack.pop())
                    else: break
                stack.push(t)

        while not stack.is_empty():
            output.enqueue(stack.pop())

        res_str=""
        while not output.is_empty():
            res_str += output.dequeue() + " "
        return res_str.strip()

    def calculate(self, expression):
        postfix = self.to_postfix(expression).split()
        stack = Stack()

        for t in postfix:
            # Унарный минус
            if t == 'u-':
                a = stack.pop()
                stack.push(-a)
            # Число => float => помещаем в стек
            elif t[0].isdigit() or '.' in t:
                stack.push(float(t))
            elif t in self.ops:
                b, a = stack.pop(), stack.pop()
                if t == '+': stack.push(a + b)
                elif t == '-': stack.push(a - b)
                elif t == '*': stack.push(a * b)
                elif t == '/': stack.push(a / b)
                elif t == '**': stack.push(a ** b)
            elif t in self.funcs:
                a = stack.pop()
                if t == 'sin': stack.push(math.sin(math.radians(a)))
                elif t == 'cos': stack.push(math.cos(math.radians(a)))
                elif t == 'tg': stack.push(math.tan(math.radians(a)))
                elif t == 'ctg': stack.push(1 / math.tan(math.radians(a)))
                elif t == 'sqrt': stack.push(math.sqrt(a))
        return stack.pop()

import math

# Шаблоны функций аффинных преобразований с лекции
def translate(points, tx, ty):
    return [(x + tx, y + ty) for x, y in points]

def rotate(points, angle_deg):
    a = math.radians(angle_deg)
    cos_a, sin_a = math.cos(a), math.sin(a)
    return [(x * cos_a - y * sin_a, x * sin_a + y * cos_a) for x, y in points]

def scale(points, sx, sy):
    return [(x * sx, y * sy) for x, y in points]

# --- Выполнение цепочки преобразований согласно условию ---
# Исходный квадрат
square = [(0, 0), (2, 0), (2, 2), (0, 2)]

# 1. Масштабирование в 1.5 раза
scaled = scale(square, 1.5, 1.5)

# 2. Поворот на 45 градусов вокруг начала координат
rotated = rotate(scaled, 45)

# 3. Сдвиг (перенос) на вектор (5, 3)
final_result = translate(rotated, 5, 3)

# Округление результатов для наглядности вывода
def round_pts(pts):
    return [(round(x, 4), round(y, 4)) for x, y in pts]

print("1. Исходный квадрат:", square)
print("2. После масштабирования:", round_pts(scaled))
print("3. После поворота (промежуточные):", round_pts(rotated))
print("4. Конечные точки (после сдвига):", round_pts(final_result))

# Определение кросс-произведения с лекции
def cross2(o, a, b):
    return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])

# Шаблон алгоритма Джарвиса с лекции (с исправлением инициализации q)
def jarvis_march(points):
    n = len(points)
    if n < 3: return points[:]
    
    # Самая левая точка
    start = min(range(n), key=lambda i: (points[i][0], points[i][1]))
    hull = []
    p = start
    
    while True:
        hull.append(points[p])
        # Чтобы избежать q == p при инициализации:
        q = (p + 1) % n
        if q == p: 
            q = (p + 2) % n
            
        for i in range(n):
            if cross2(points[p], points[i], points[q]) > 0:
                q = i
        p = q
        if p == start:
            break
    return hull

# Генерируем 10 точек, включая гарантированную коллинеарность на одной линии
pts = [(0, 0), (2, 0), (4, 0), (4, 2), (4, 4), (2, 4), (0, 4), (0, 2), (1, 1), (2, 2)]
print("Набор из 10 точек:", pts)
print("Выпуклая оболочка (Джарвис):", jarvis_march(pts))






import math
import matplotlib.pyplot as plt

def cross2(o, a, b):
    return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])

# Скорректированный шаблон Грэхема с лекции
def graham_scan(points):
    if len(points) < 3: return points[:]
    
    # Нижняя (затем левая) точка
    start = min(points, key=lambda p: (p[1], p[0]))
    
    def angle_key(p):
        if p == start:
            return (-math.inf, 0) # Гарантируем, что start будет самым первым
        return (math.atan2(p[1] - start[1], p[0] - start[0]), 
                (p[0] - start[0])**2 + (p[1] - start[1])**2)
                
    sorted_pts = sorted(points, key=angle_key)
    
    stack = []
    for p in sorted_pts:
        while len(stack) >= 2 and cross2(stack[-2], stack[-1], p) <= 0:
            stack.pop() # Убираем правый поворот и коллинеарные внутренние точки
        stack.append(p)
    return stack

# Тот же самый набор из 10 точек для сравнения
pts = [(0, 0), (2, 0), (4, 0), (4, 2), (4, 4), (2, 4), (0, 4), (0, 2), (1, 1), (2, 2)]
