class Node:
    def __init__(self, key):
        self.key = key
        self.left = None
        self.right = None

class BST:
    def __init__(self):
        self.root = None
        self.comp_count = 0  # Счетчик для задания №2

    # --- МЕТОДЫ ВСТАВКИ (Слайд 1) ---
    def insert(self, key):
        self.root = self._insert(self.root, key)

    def _insert(self, node, key):
        if node is None:
            return Node(key)        # нашли место - создаём узел
        if key < node.key:
            node.left = self._insert(node.left, key)
        elif key > node.key:
            node.right = self._insert(node.right, key)
        return node                 # дубликаты игнорируем

    # --- МЕТОД ПОИСКА (Слайд 2) ---
    def search(self, key):
        self.comp_count = 0
        return self._search(self.root, key)

    def _search(self, node, key):
        if node is None:
            return None             # не найден
        self.comp_count += 1        # считаем каждое сравнение
        if key == node.key:
            return node             # найден!
        if key < node.key:
            return self._search(node.left, key)
        else:
            return self._search(node.right, key)

    # --- МЕТОДЫ ОБХОДА (Слайд 3) ---
    def inorder(self, node):
        if node:
            self.inorder(node.left)
            print(node.key, end=' ')
            self.inorder(node.right)

    def preorder(self, node):
        if node:
            print(node.key, end=' ')
            self.preorder(node.left)
            self.preorder(node.right)

    def postorder(self, node):
        if node:
            self.postorder(node.left)
            self.postorder(node.right)
            print(node.key, end=' ')

    # Вспомогательный метод для замера высоты
    def get_height(self, node):
        if node is None:
            return 0
        return 1 + max(self.get_height(node.left), self.get_height(node.right))

# --- ВЫПОЛНЕНИЕ ЗАДАНИЯ ПО ПУНКТАМ ---

# Ключи из ТЗ
keys_list = [50, 30, 120, 70, 20, 90, 40, 60, 80]

# --- Пункт №1: Построение BST и обходы ---
tree1 = BST()
for k in keys_list:
    tree1.insert(k)

print("=== ЗАДАНИЕ №1 ===")
print("Обходы дерева (построено из исходного списка):")
print("In-order:  ", end=""); tree1.inorder(tree1.root); print()
print("Pre-order: ", end=""); tree1.preorder(tree1.root); print()
print("Post-order:", end=""); tree1.postorder(tree1.root); print()

# --- Пункт №2: Сравнение с отсортированным списком ---
print("\n=== ЗАДАНИЕ №2 ===")
tree2 = BST()
# Строим дерево из отсортированного списка (худший случай)
for k in sorted(keys_list):
    tree2.insert(k)

print(f"Высота дерева №1 (обычное): {tree1.get_height(tree1.root)}")
print(f"Высота дерева №2 (отсортированное): {tree2.get_height(tree2.root)}")

# Сравнение поиска
target_key = 80
tree1.search(target_key)
c1 = tree1.comp_count

tree2.search(target_key)
c2 = tree2.comp_count

print(f"Сравнений для поиска '{target_key}' в дереве №1: {c1}")
print(f"Сравнений для поиска '{target_key}' в дереве №2: {c2}")

# --- Пункт №3: Обоснование сложности ---
print("\n=== ЗАДАНИЕ №3 ===")
print("Сложность BST:")
print("- Средний случай: O(log n) — если дерево сбалансировано (как наше дерево №1).")
print("- Худший случай: O(n) — если дерево вырождается в список (как наше дерево №2).")
print("- Лучший случай: O(1) — если искомый ключ находится в корне.")
