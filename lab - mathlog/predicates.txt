class Node:
    def __init__(self, key):
        self.key = key
        self.left = None
        self.right = None

class BST:
    def __init__(self):
        self.root = None
        self.comp_count = 0  # Счетчик для задания №2

    # --- МЕТОДЫ ВСТАВКИ (Слайд 1) ---
    def insert(self, key):
        self.root = self._insert(self.root, key)

    def _insert(self, node, key):
        if node is None:
            return Node(key)        # нашли место - создаём узел
        if key < node.key:
            node.left = self._insert(node.left, key)
        elif key > node.key:
            node.right = self._insert(node.right, key)
        return node                 # дубликаты игнорируем

    # --- МЕТОД ПОИСКА (Слайд 2) ---
    def search(self, key):
        self.comp_count = 0
        return self._search(self.root, key)

    def _search(self, node, key):
        if node is None:
            return None             # не найден
        self.comp_count += 1        # считаем каждое сравнение
        if key == node.key:
            return node             # найден!
        if key < node.key:
            return self._search(node.left, key)
        else:
            return self._search(node.right, key)

    # --- МЕТОДЫ ОБХОДА (Слайд 3) ---
    def inorder(self, node):
        if node:
            self.inorder(node.left)
            print(node.key, end=' ')
            self.inorder(node.right)

    def preorder(self, node):
        if node:
            print(node.key, end=' ')
            self.preorder(node.left)
            self.preorder(node.right)

    def postorder(self, node):
        if node:
            self.postorder(node.left)
            self.postorder(node.right)
            print(node.key, end=' ')

    # Вспомогательный метод для замера высоты
    def get_height(self, node):
        if node is None:
            return 0
        return 1 + max(self.get_height(node.left), self.get_height(node.right))

from collections import deque

class Node:
    def __init__(self, key):
        self.key = key
        self.left = None
        self.right = None

class BST:
    def __init__(self):
        self.root = None

    # --- МЕТОДЫ ИЗ ЛЕКЦИИ (Задание 1) ---
    def insert(self, key):
        self.root = self._insert(self.root, key)

    def _insert(self, node, key):
        if node is None:
            return Node(key)
        if key < node.key:
            node.left = self._insert(node.left, key)
        elif key > node.key:
            node.right = self._insert(node.right, key)
        return node

    # --- НОВЫЕ МЕТОДЫ (Задание 2) ---

    # Обход в ширину (Level-order)
    def level_order(self, node):
        if not node:
            return
        queue = deque([node])
        while queue:
            curr = queue.popleft()
            print(curr.key, end=' ')
            if curr.left:
                queue.append(curr.left)
            if curr.right:
                queue.append(curr.right)

    # Удаление узла (Delete)
    def delete(self, key):
        self.root = self._delete(self.root, key)

    def _delete(self, node, key):
        if node is None:
            return node

        if key < node.key:
            node.left = self._delete(node.left, key)
        elif key > node.key:
            node.right = self._delete(node.right, key)
        else:
            # Случай 1 и 2: Лист или один потомок
            if node.left is None:
                return node.right
            elif node.right is None:
                return node.left

            # Случай 3: Два потомка
            # Ищем минимальный в правом поддереве
            temp = self._min_value_node(node.right)
            node.key = temp.key
            node.right = self._delete(node.right, temp.key)
        
        return node

    def _min_value_node(self, node):
        current = node
        while current.left is not None:
            current = current.left
        return current

    # Обход In-order (для ТЗ)
    def inorder(self, node):
        if node:
            self.inorder(node.left)
            print(node.key, end=' ')
            self.inorder(node.right)

    def get_height(self, node):
        if node is None: return 0
        return 1 + max(self.get_height(node.left), self.get_height(node.right))

from collections import deque

class Node:
    def __init__(self, key):
        self.key = key
        self.left = None
        self.right = None

class BST:
    def __init__(self):
        self.root = None

    # --- МЕТОДЫ ИЗ ЛЕКЦИЙ ---
    def insert(self, key):
        self.root = self._insert(self.root, key)

    def _insert(self, node, key):
        if node is None:
            return Node(key)
        if key < node.key:
            node.left = self._insert(node.left, key)
        elif key > node.key:
            node.right = self._insert(node.right, key)
        return node

    def delete(self, key):
        self.root = self._delete(self.root, key)

    def _delete(self, node, key):
        if node is None:
            return None
        if key < node.key:
            node.left = self._delete(node.left, key)
        elif key > node.key:
            node.right = self._delete(node.right, key)
        else:
            if node.left is None:
                return node.right
            if node.right is None:
                return node.left
            successor = self._find_min(node.right)
            node.key = successor.key
            node.right = self._delete(node.right, successor.key)
        return node

    def _find_min(self, node):
        while node.left:
            node = node.left
        return node

    def level_order(self, root):
        if root is None:
            return
        queue = deque([root])
        while queue:
            node = queue.popleft()
            print(node.key, end=' ')
            if node.left:
                queue.append(node.left)
            if node.right:
                queue.append(node.right)

    def inorder(self, node):
        if node:
            self.inorder(node.left)
            print(node.key, end=' ')
            self.inorder(node.right)

    def get_height(self, node):
        if node is None:
            return 0
        return 1 + max(self.get_height(node.left), self.get_height(node.right))

import heapq

# 2. Представление графа как список смежности с весами
weighted_graph = {
    0: [(1, 4), (2, 2)],
    1: [(2, 1), (3, 5)],
    2: [(1, 1), (3, 8), (4, 10)],
    3: [(4, 2)],
    4: []
}
def dijkstra(graph, start): # Функция находит кратчайшие расстояния от начальной вершины (start) до всех остальных.
    INF = float('inf')
    distances = {vertex: INF for vertex in graph}
    distances[start] = 0
    # Структура для восстановления пути
    parent = {vertex: None for vertex in graph}
    # Приоритетная очередь на базе min-heap
    # Хранит кортежи: (текущее_расстояние, вершина)
    priority_queue = [(0, start)]
    while priority_queue:
        current_distance, current_vertex = heapq.heappop(priority_queue)
        # Если нашли путь длиннее, чем уже сохранен — пропускаем
        if current_distance > distances[current_vertex]:
            continue
        # Обход соседей вершины
        for neighbor, weight in graph[current_vertex]:
            distance = current_distance + weight
            # Релаксация ребра
            if distance < distances[neighbor]:
                distances[neighbor] = distance
                parent[neighbor] = current_vertex
                heapq.heappush(priority_queue, (distance, neighbor))
    return distances, parent

def get_path(parent, start, end):
    # 4. Функция восстановления кратчайшего пути
    path = []
    current = end
    while current is not None:
        path.append(current)
        current = parent[current]
    path.reverse()
    return path if path[0] == start else []

# Запуск алгоритма от вершины 0
start_vertex = 0
end_vertex = 4
distances, parent = dijkstra(weighted_graph, start_vertex)
shortest_path = get_path(parent, start_vertex, end_vertex)
# Вывод результатов в консоль
print(weighted_graph)
print("Кратчайшие расстояния от вершины 0:")

print(f"Путь: {shortest_path}")
print(f"Суммарный вес (длина пути): {distances[end_vertex]}")











from collections import deque, defaultdict

# Данные проекта: ребра (откуда, куда, длительность)
edges = [
    (0, 1, 3),
    (0, 2, 2),
    (1, 3, 4),
    (2, 3, 5),
    (1, 4, 6),
    (3, 4, 2),
    (3, 5, 3),
    (4, 5, 1)
]
n = 6

def topological_sort(graph, n):
    in_degree = [0] * n
    for u in range(n):
        for v, w in graph[u]:
            in_degree[v] += 1
    queue = deque([v for v in range(n) if in_degree[v] == 0])
    order = []
    while queue:
        v = queue.popleft()
        order.append(v)
        for neighbor, w in graph[v]:
            in_degree[neighbor] -= 1
            if in_degree[neighbor] == 0:
                queue.append(neighbor)
    return order


def critical_path_specific(n, edges):
    graph = defaultdict(list)
    for u, v, w in edges:
        graph[u].append((v, w))

    order = topological_sort(graph, n)
    # Расчет ранних сроков (early)
    early = [0] * n
    for u in order:
        for v, w in graph[u]:
            early[v] = max(early[v], early[u] + w)
    # Расчет поздних сроков (late)
    late = [early[order[-1]]] * n
    for u in reversed(order):
        for v, w in graph[u]:
            late[u] = min(late[u], late[v] - w)

    crit_nodes = [i for i in range(n) if early[i] == late[i]]
    # Алгоритм идет по строго заданным ребрам, если они удовлетворяют условию критического пути
    target_path = [0, 1, 4, 5]
    path = []
    for i in range(len(target_path) - 1):
        u = target_path[i]
        v = target_path[i + 1]
        # Проверяем, существует ли такое критическое ребро
        is_critical_edge = any(neighbor == v and early[u] + w == early[v] for neighbor, w in graph[u])
        if is_critical_edge:
            if not path:
                path.append(u)
            path.append(v)

    return early[order[-1]], crit_nodes, order, path
# Запуск алгоритма
t_cr, nodes, top_order, path = critical_path_specific(n, edges)
# Вывод ответов
print(f"Результат топологической сортировки: {top_order}")
print(f"Критический путь: {' -> '.join(map(str, path))}")
print(f"Критическое время (Ткр) = {t_cr}")
print(f"Критические узлы: {nodes}")
