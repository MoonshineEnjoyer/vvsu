class Node:
    def __init__(self, key):
        self.key = key
        self.left = None
        self.right = None

class BST:
    def __init__(self):
        self.root = None
        self.comp_count = 0 # счетчик для пункта №2

    # --- МЕТОДЫ ВСТАВКИ (Слайд 1) ---
    def insert(self, key):
        self.root = self._insert(self.root, key)

    def _insert(self, node, key):
        if node is None:
            return Node(key)        # нашли место - создаём узел
        if key < node.key:
            node.left = self._insert(node.left, key)
        elif key > node.key:
            node.right = self._insert(node.right, key)
        return node                 # дубликаты игнорируем

    # --- МЕТОД ПОИСКА (Слайд 2) ---
    def search(self, key):
        self.comp_count = 0
        return self._search(self.root, key)

    def _search(self, node, key):
        if node is None:
            return None             # не найден
        self.comp_count += 1
        if key == node.key:
            return node             # найден!
        if key < node.key:
            return self._search(node.left, key)
        else:
            return self._search(node.right, key)

    # --- МЕТОДЫ ОБХОДА (Слайд 3) ---
    def inorder(self, node):
        if node:
            self.inorder(node.left)
            print(node.key, end=' ')
            self.inorder(node.right)

    def preorder(self, node):
        if node:
            print(node.key, end=' ')
            self.preorder(node.left)
            self.preorder(node.right)

    def postorder(self, node):
        if node:
            self.postorder(node.left)
            self.postorder(node.right)
            print(node.key, end=' ')

    # Метод для замера высоты (Пункт 2 ТЗ)
    def get_height(self, node):
        if node is None:
            return 0
        return 1 + max(self.get_height(node.left), self.get_height(node.right))

# --- ВЫПОЛНЕНИЕ ЗАДАНИЯ ---

keys = [50, 30, 120, 70, 20, 90, 40, 60, 80]

# 1. Построение и вывод обходов
tree = BST()
for k in keys:
    tree.insert(k)

print("Задание 1: Результаты обходов")
print("In-order:  ", end=''); tree.inorder(tree.root); print()
print("Pre-order: ", end=''); tree.preorder(tree.root); print()
print("Post-order:", end=''); tree.postorder(tree.root); print()

# 2. Сравнение высоты и поиска
print("\nЗадание 2: Сравнение")
tree_sorted = BST()
for k in sorted(keys):
    tree_sorted.insert(k)

print(f"Высота обычного дерева: {tree.get_height(tree.root)}")
print(f"Высота отсортированного: {tree_sorted.get_height(tree_sorted.root)}")

tree.search(80)
c1 = tree.comp_count
tree_sorted.search(80)
c2 = tree_sorted.comp_count
print(f"Сравнений при поиске '80': в обычном = {c1}, в отсортированном = {c2}")
